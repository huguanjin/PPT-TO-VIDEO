        if start <= code <= end:
            return weight
    
    # 2. 使用Unicode分类
    category = unicodedata.category(char)
    if category in cls.CATEGORY_WEIGHTS:
        return cls.CATEGORY_WEIGHTS[category]
    
    # 3. 默认权重
    return 1.0
