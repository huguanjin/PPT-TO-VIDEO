```python
class EnhancedSubtitleConfigLoader:
    def _migrate_legacy_config(self):
        """迁移传统配置到预设系统"""
        complexity_score = 0
        # 分析算法配置
        if self.config.get('use_dp_algorithm'): complexity_score += 2
        if self.config.get('use_spacy'): complexity_score += 3
        if self.config.get('ai_splitting'): complexity_score += 2
        
        # 根据复杂度推荐预设
        if complexity_score <= 3: suggested_preset = 'simple'
        elif complexity_score <= 6: suggested_preset = 'standard'  
        else: suggested_preset = 'professional'
```

- **预设切换**: 支持运行时动态切换预设

```python
class SmartSubtitleConfigLoader:
    def _ensure_backward_compatibility(self):
        """确保向后兼容性"""
        if "smart_subtitle_processing" not in self.config:
            self.config["smart_subtitle_processing"] = {}
        
        # 映射预设配置到传统结构
        subtitle_config = self.config["smart_subtitle_processing"]
        subtitle_config.update({
            "enabled": True,
            "max_length": self.config.get("max_length", 75),
            "use_ai_splitting": self.config.get("ai_splitting", False),
            "use_dp_algorithm": self.config.get("use_dp_algorithm", True),
            "use_spacy": self.config.get("use_spacy", True)
        })
```

```python
def validate_config(config: Dict[str, Any], preset: str) -> Dict[str, Any]:
    """配置验证"""
    errors = []
    warnings = []
    
    # 基础验证
    if config.get("max_length", 0) <= 0:
        errors.append("max_length必须是正数")
    
    # 预设特定验证
    if preset == "professional" and config.get("max_length", 100) > 50:
        warnings.append("专业模式建议使用较短的字幕长度")
    
    return {
        "valid": len(errors) == 0,
        "errors": errors,
        "warnings": warnings
    }
```

## 使用示例

### 1. 基础使用

```python

simple_config = loader.get_simple_config()
```

### 2. 预设管理

```python
# 获取可用预设

# 切换预设（保留自定义配置）
loader.switch_preset('professional', keep_overrides=True)

info = loader.get_preset_info()
```
